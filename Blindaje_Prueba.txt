CREATE OR REPLACE FUNCTION aplicar_transaccion() RETURNS TRIGGER AS $$
DECLARE
    v_socio_cuenta   INTEGER;
    v_socio_prestamo INTEGER;
    v_saldo          NUMERIC(12,2);
BEGIN
    PERFORM set_config('caja.ledger', 'on', true);

    IF NEW.prestamo_id IS NOT NULL THEN
        SELECT socio_id INTO v_socio_cuenta   FROM cuentas_ahorro WHERE id = NEW.cuenta_id;
        SELECT socio_id INTO v_socio_prestamo FROM prestamos      WHERE id = NEW.prestamo_id;
        IF v_socio_cuenta IS DISTINCT FROM v_socio_prestamo THEN
            RAISE EXCEPTION 'La cuenta (socio %) y el prestamo (socio %) no coinciden',
                            v_socio_cuenta, v_socio_prestamo;
        END IF;
    END IF;

    IF NEW.tipo IN ('abono_prestamo','penalizacion','desembolso_prestamo')
       AND NEW.prestamo_id IS NULL THEN
        RAISE EXCEPTION 'La transaccion % requiere prestamo_id', NEW.tipo;
    END IF;

    IF NEW.tipo IN ('deposito','interes_ahorro')
       AND (NEW.monto < 100.00 OR NEW.monto > 10000.00) THEN
        RAISE EXCEPTION 'Monto de % fuera de rango 100-10000: %', NEW.tipo, NEW.monto;
    END IF;

    IF NEW.tipo IN ('deposito','interes_ahorro','apertura','desembolso_prestamo') THEN
        UPDATE cuentas_ahorro SET saldo = saldo + NEW.monto WHERE id = NEW.cuenta_id;
    ELSIF NEW.tipo = 'retiro' THEN
        SELECT saldo INTO v_saldo FROM cuentas_ahorro WHERE id = NEW.cuenta_id FOR UPDATE;
        IF NEW.monto > v_saldo THEN
            RAISE EXCEPTION 'Fondos insuficientes: saldo % < retiro %', v_saldo, NEW.monto;
        END IF;
        UPDATE cuentas_ahorro SET saldo = saldo - NEW.monto WHERE id = NEW.cuenta_id;
    ELSIF NEW.tipo = 'abono_prestamo' THEN
        UPDATE prestamos SET saldo_pendiente = saldo_pendiente - NEW.monto WHERE id = NEW.prestamo_id;
    ELSIF NEW.tipo = 'penalizacion' THEN
        UPDATE prestamos SET saldo_pendiente = saldo_pendiente + NEW.monto WHERE id = NEW.prestamo_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_aplicar_transaccion ON transacciones;
CREATE TRIGGER trg_aplicar_transaccion
    BEFORE INSERT ON transacciones
    FOR EACH ROW EXECUTE FUNCTION aplicar_transaccion();

CREATE OR REPLACE FUNCTION proteger_saldo_directo() RETURNS TRIGGER AS $$
BEGIN
    IF NEW.saldo <> OLD.saldo
       AND current_setting('caja.ledger', true) IS DISTINCT FROM 'on' THEN
        RAISE EXCEPTION 'El saldo solo puede modificarse via transacciones';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_proteger_saldo ON cuentas_ahorro;
CREATE TRIGGER trg_proteger_saldo
    BEFORE UPDATE OF saldo ON cuentas_ahorro
    FOR EACH ROW EXECUTE FUNCTION proteger_saldo_directo();
